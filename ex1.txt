sim_p1 = function(n=100) {
  M = rnorm(n)
  E <- rnorm(n, M)
  CI = rnorm(n, M + E)
  WL <- rnorm(n, E + CI)
  # correct: control for CI to get direct effect
  correct = coef(lm(WL ~ E + CI))['E']
  # wrong: only looking at E gives total-ish effect, not the direct one
  wrong = coef(lm(WL ~ E))['E']
  return(c(correct, wrong))
}
res1 <- replicate(1000, sim_p1())
plot(density(res1[1,]), lwd=2, main="prob 1: direct effect E -> WL", xlab="posterior mean", x
lim=c(0, 3))
lines(density(res1[2,]), lwd=2, col=2)
legend("topright", legend=c("correct (WL ~ E + CI)", "wrong (WL ~ E)"), col=c(1, 2), lwd=2)

f2 = function(n=100) {
  CE <- rnorm(n)
  EL = rnorm(n, CE)
  F_var <- rnorm(n, EL)
  JS = rnorm(n, CE + F_var + EL)
  # correct: control CE only for total effect
  correct <- coef(lm(JS ~ EL + CE))['EL']
  # wrong: controling for F blocks part of total effect
  wrong = coef(lm(JS ~ EL + CE + F_var))['EL']
  return(c(correct, wrong))
}
sim_2 = replicate(1000, f2())

func_3 <- function(n=100) {
  A = rnorm(n)
  SI <- rnorm(n, A)
  M = rnorm(n, SI)
  RR <- rnorm(n, SI + A + M)
  # model with SI
  model_si <- coef(lm(RR ~ M + SI))['M']
  # model with SI and Age
  model_age = coef(lm(RR ~ M + SI + A))['M']
  return(c(model_si, model_age))
}

sim_p4 = function(n=100) {
  EC <- rnorm(n)
  B = rnorm(n, EC)
  Q <- rnorm(n)
  A = rnorm(n, B + Q)
  S <- rnorm(n, A + Q + EC)
  # correct: control Q and EC
  correct = coef(lm(S ~ A + Q + EC))['A']
  # wrong: forget to control Q
  wrong <- coef(lm(S ~ A + EC))['A']
  return(c(correct, wrong))
}
res4 = replicate(1000, sim_p4())
plot(density(res4[1,]), lwd=2, main="prob 4: total effect A -> S", xlab="posterior mean")
lines(density(res4[2,]), lwd=2, col=2)
legend("topleft", legend=c("correct", "wrong"), col=c(1, 2), lwd=2)
